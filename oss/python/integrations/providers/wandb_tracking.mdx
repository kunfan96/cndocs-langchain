```python
"""主函数。
```

```
# WandbCallbackHandler.flush_tracker(...) 的默认值

reset: bool = True,

```

`flush_tracker` 函数用于将 LangChain 会话记录到 Weights & Biases。它接收 LangChain 模块或代理，并至少将提示和生成内容以及 LangChain 模块的序列化形式记录到指定的 Weights & Biases 项目。默认情况下，我们重置会话，而不是直接结束会话。

### 在链中使用

```python
from langchain_classic.chains import LLMChain
from langchain_core.prompts import PromptTemplate
```

```python
# 场景 2 - 链
template = """你是一名剧作家。根据给定的戏剧标题，你的任务是为该标题写一个剧情简介。
标题：{title}
剧作家：这是上述戏剧的剧情简介："""
prompt_template = PromptTemplate(input_variables=["title"], template=template)
synopsis_chain = LLMChain(llm=llm, prompt=prompt_template, callbacks=callbacks)

test_prompts = [
    {
        "title": "关于突破游戏设计界限的优秀电子游戏的纪录片"
    },
    {"title": "可卡因熊对决海洛因狼"},
    {"title": "同类最佳的 MLOps 工具"},
]
synopsis_chain.apply(test_prompts)
wandb_callback.flush_tracker(synopsis_chain, name="agent")
```

### 与代理一起使用

```python
from langchain.agents import create_agent, load_tools
```

```python
# 场景 3 - 带工具的代理
tools = load_tools(["serpapi", "llm-math"], llm=llm)

agent = create_agent(
    model=llm,
    tools=tools,
    callbacks=callbacks,
)

agent.invoke(
    "莱昂纳多·迪卡普里奥的女朋友是谁？她现在的年龄的 0.43 次方是多少？"
)

wandb_callback.flush_tracker(agent, reset=False, finish=True)

```
